#### 1.2. SprintService — Business logic
- File: `backend/src/sprints/sprints.service.ts`

```typescript
@Injectable()
export class SprintsService {
  constructor(private prisma: PrismaService) {}

  async create(dto: CreateSprintDto, actor: CurrentActor) {
    // 1. Validate project access
    await this.validateProjectAccess(dto.projectId, actor, 'PM');
    // 2. Check no ACTIVE sprint exists in project
    const activeSprint = await this.prisma.sprint.findFirst({
      where: { projectId: dto.projectId, status: 'ACTIVE' },
    });
    if (activeSprint) {
      throw new ConflictException('Project đang có sprint ACTIVE. Complete sprint hiện tại trước.');
    }
    // 3. Auto-calculate dates if not provided
    const startDate = dto.startDate ?? new Date();
    const endDate = dto.endDate ?? new Date(startDate.getTime() + dto.durationDays * 24 * 60 * 60 * 1000);
    // 4. Create
    return this.prisma.sprint.create({
      data: {
        ...dto,
        startDate,
        endDate,
        createdById: actor.userId,
        status: 'PLANNING',
      },
    });
  }

  async start(id: string, actor: CurrentActor) {
    const sprint = await this.findOne(id, actor);
    if (sprint.status !== 'PLANNING') {
      throw new BadRequestException(`Sprint phải ở trạng thái PLANNING. Hiện tại: ${sprint.status}`);
    }
    // 1. Check no other ACTIVE sprint in same project
    const otherActive = await this.prisma.sprint.findFirst({
      where: { projectId: sprint.projectId, status: 'ACTIVE', id: { not: id } },
    });
    if (otherActive) {
      throw new ConflictException('Project đã có sprint ACTIVE khác.');
    }
    // 2. Update status
    return this.prisma.sprint.update({
      where: { id },
      data: {
        status: 'ACTIVE',
        startDate: sprint.startDate ?? new Date(),
      },
    });
  }

  async complete(id: string, actor: CurrentActor) {
    const sprint = await this.findOne(id, actor);
    if (sprint.status !== 'ACTIVE') {
      throw new BadRequestException(`Sprint phải ở trạng thái ACTIVE. Hiện tại: ${sprint.status}`);
    }
    // 1. Update sprint status
    const updated = await this.prisma.sprint.update({
      where: { id },
      data: { status: 'COMPLETED', completedAt: new Date() },
    });
    // 2. Optional: Move incomplete tasks back to backlog
    // (decide based on product requirement — for now keep as-is)
    return updated;
  }

  async getBurndown(id: string, actor: CurrentActor) {
    const sprint = await this.findOne(id, actor);
    if (sprint.status === 'PLANNING') {
      return { ideal: [], actual: [], message: 'Sprint chưa start' };
    }
    // 1. Get all tasks in sprint
    const tasks = await this.prisma.task.findMany({
      where: { sprintId: id, deletedAt: null },
      select: { id: true, storyPoints: true, status: true, updatedAt: true },
    });
    // 2. Calculate burndown points per day
    const totalPoints = tasks.reduce((sum, t) => sum + (t.storyPoints ?? 0), 0);
    const sprintDays = sprint.durationDays;
    const burndown: Array<{ day: string; ideal: number; actual: number; date: string }> = [];
    // ... calculate ideal line (linear) and actual line (based on completed tasks per day)
    return burndown;
  }
}
```

```typescript
async create(dto: CreateQuickTaskDto, actor: CurrentActor) {
  // 1. Validate assignee based on actor role
  if (actor.role === 'DEPARTMENT_HEAD') {
    // DEPT_HEAD chỉ giao được cho nhân viên cùng phòng
    const assignee = await this.prisma.user.findUnique({
      where: { id: dto.assigneeId },
      select: { departmentId: true, status: true },
    });
    if (!assignee) throw new NotFoundException('Assignee không tồn tại');
    if (assignee.status !== 'ACTIVE') throw new BadRequestException('Nhân viên đã bị vô hiệu hóa');
    
    // Check department: actor phải là DEPT_HEAD của department này
    const actorUser = await this.prisma.user.findUnique({
      where: { id: actor.userId },
      select: { departmentId: true, role: true },
    });
    if (actorUser?.departmentId !== assignee.departmentId) {
      throw new ForbiddenException('DEPT_HEAD chỉ giao task cho nhân viên cùng phòng ban');
    }
  }
  
  if (actor.role === 'PM') {
    // PM chỉ giao được cho thành viên trong project (nếu có projectId)
    if (dto.projectId) {
      const member = await this.prisma.projectMember.findUnique({
        where: { projectId_userId: { projectId: dto.projectId, userId: dto.assigneeId } },
      });
      if (!member) throw new ForbiddenException('Assignee không thuộc project này');
    }
  }
  
  // 2. Create quick task
  const quickTask = await this.prisma.quickTask.create({
    data: {
      title: dto.title,
      description: dto.description,
      projectId: dto.projectId,
      assigneeId: dto.assigneeId,
      assignedById: actor.userId,
      dueDate: dto.dueDate ?? endOfDay(new Date()),  // default: cuối ngày hôm nay
      priority: dto.priority ?? 'MEDIUM',
      status: 'TODO',
    },
  });
  
  // 3. Send notification to assignee
  await this.notificationService.send(quickTask.assigneeId, {
    type: 'TASK_ASSIGNED',
    title: 'Bạn có việc mới',
    message: `${actor.name} đã giao: ${quickTask.title}`,
    link: `/quick-tasks/${quickTask.id}`,
  });
  
  return quickTask;
}
```

#### 6.2. DepartmentDashboardService (Backend)

```typescript
async getDepartmentDashboard(userId: string): Promise<DepartmentDashboardDto> {
  // 1. Get user's department
  const user = await this.prisma.user.findUnique({
    where: { id: userId },
    select: { departmentId: true, role: true, name: true },
  });
  if (!user?.departmentId) throw new ForbiddenException('Bạn không thuộc phòng ban nào');
  
  // 2. Get department members
  const members = await this.prisma.user.findMany({
    where: { departmentId: user.departmentId, status: 'ACTIVE', deletedAt: null },
    select: { id: true, name: true, avatar: true, position: true },
  });
  
  // 3. Today tasks
  const todayStart = startOfDay(new Date());
  const todayEnd = endOfDay(new Date());
  const memberIds = members.map(m => m.id);
  
  // Tasks completed today
  const doneToday = await this.prisma.task.count({
    where: {
      assigneeId: { in: memberIds },
      status: 'DONE',
      updatedAt: { gte: todayStart, lte: todayEnd },
      deletedAt: null,
    },
  });
  
  // 4. Overdue tasks
  const overdueTasks = await this.prisma.task.count({
    where: {
      assigneeId: { in: memberIds },
      status: { not: 'DONE' },
      dueDate: { lt: new Date() },
      deletedAt: null,
    },
  });
  
  // 5. Late check-ins today
  const lateCheckIns = await this.prisma.attendanceRecord.count({
    where: {
      userId: { in: memberIds },
      date: todayStart,
      status: { in: ['LATE', 'VERY_LATE'] },
    },
  });
  
  // 6. Absent today
  const presentToday = await this.prisma.attendanceRecord.count({
    where: { userId: { in: memberIds }, date: todayStart, checkInTime: { not: null } },
  });
  const absentToday = members.length - presentToday;
  
  // 7. Team activity feed
  const recentActivities = await this.prisma.task.findMany({
    where: { assigneeId: { in: memberIds }, deletedAt: null },
    include: {
      assignee: { select: { name: true, avatar: true } },
      project: { select: { name: true, color: true } },
    },
    orderBy: { updatedAt: 'desc' },
    take: 10,
  });
  
  // 8. Task list by member
  const memberTasks = await Promise.all(
    members.map(async (m) => {
      const tasks = await this.prisma.task.findMany({
        where: { assigneeId: m.id, deletedAt: null },
        select: { id: true, status: true, dueDate: true, priority: true },
      });
      const total = tasks.length;
      const done = tasks.filter(t => t.status === 'DONE').length;
      const overdue = tasks.filter(t => t.status !== 'DONE' && t.dueDate && t.dueDate < new Date()).length;
      return {
        userId: m.id,
        userName: m.name,
        avatar: m.avatar,
        position: m.position,
        totalTasks: total,
        doneTasks: done,
        overdueTasks: overdue,
        progressPercent: total > 0 ? Math.round((done / total) * 100) : 0,
      };
    })
  );
  
  return {
    departmentName: '...',  // query from Department table
    totalMembers: members.length,
    completedToday: doneToday,
    overdueTasks,
    lateToday: lateCheckIns,
    absentToday,
    recentActivities: recentActivities.map(/* map to DTO */),
    memberTasks,
  };
}
```
